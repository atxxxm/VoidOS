# Вывести содержимое текущей директории  
ls

# Вывести содержимое нескольких директорий  
ls dir1/ dir2/

# Показать скрытые файлы  
ls -a

# Подробный вывод (размер, время изменения, тип)  
ls -l

# Подробный вывод со скрытыми файлами  
ls -la

# Вывести содержимое указанной директории  
ls /path/to/folder

# Комбинированный вывод: подробно, со скрытыми файлами, из нескольких путей  
ls -l -a ~/docs ~/projects
