# Move a file  
mv file.txt ./backup/

# Rename a file  
mv old_name.txt new_name.txt

# Move a directory (recursively)  
mv -r my_folder/ archive/

# Rename a directory  
mv old_dir/ new_dir/

# Move multiple files into an existing directory  
mv file1.txt file2.log ./dest/

# Move a file, overwriting the destination if it already exists  
mv document.pdf ./final.pdf
